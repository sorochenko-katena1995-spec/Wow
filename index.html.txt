  $('#endSpin').appendChild(clone);

  /* ---------- Навигация-точки и анимация при листании ---------- */
  var invite = $('#invite'), sections = $$('#invite section'), dotsEl = $('#dots');
  sections.forEach(function(s, idx){
    var b = document.createElement('button');
    b.setAttribute('aria-label','Страница '+(idx+1));
    b.addEventListener('click', function(){ s.scrollIntoView({behavior:'smooth'}); });
    dotsEl.appendChild(b);
  });
  var dotBtns = $$('button', dotsEl);
  var io = null;
  function startObserver(){
    io = new IntersectionObserver(function(entries){
      entries.forEach(function(e){
        var idx = sections.indexOf(e.target);
        if(e.isIntersecting && e.intersectionRatio > .55){
          e.target.classList.add('in');
          dotBtns.forEach(function(d,k){ d.classList.toggle('on', k===idx); });
        } else if(!e.isIntersecting){
          e.target.classList.remove('in'); // при возвращении анимация проигрывается снова
        }
      });
    }, {root: invite, threshold: [0, .55, .8]});
    sections.forEach(function(s){ io.observe(s); });
  }

  /* ---------- Музыка (Web Audio, без внешних файлов) ---------- */
  var ac = null, master = null, timer = null, nextBar = 0, barIdx = 0, playing = false;
  var BEAT = 60/62, EIGHTH = BEAT/2, BAR = BEAT*4;
  var mtof = function(m){ return 440*Math.pow(2,(m-69)/12); };
  // Bm – G – D – A (мягкая, трогательная последовательность)
  var CHORDS = [
    {bass:47, arp:[54,59,62,66], mel:[[0,78,3],[3,76,1],[4,74,4]]},
    {bass:43, arp:[50,55,59,62], mel:[[0,74,3],[3,71,1],[4,79,4]]},
    {bass:50, arp:[57,62,66,69], mel:[[0,78,3],[3,74,1],[4,76,2],[6,78,2]]},
    {bass:45, arp:[52,57,61,64], mel:[[0,76,3],[3,73,1],[4,76,4]]}
  ];
  var ORDER = [0,1,2,1,3,2,1,2];

  function makeReverb(){
    var len = ac.sampleRate*3.2, buf = ac.createBuffer(2,len,ac.sampleRate);
    for(var ch=0; ch<2; ch++){
      var d = buf.getChannelData(ch);
      for(var i=0;i<len;i++) d[i] = (Math.random()*2-1)*Math.pow(1-i/len,2.6);
    }
    var cv = ac.createConvolver(); cv.buffer = buf; return cv;
  }
  function note(m, t, dur, vol){
    var g = ac.createGain();
    g.gain.setValueAtTime(0.0001,t);
    g.gain.exponentialRampToValueAtTime(vol,t+.012);
    g.gain.exponentialRampToValueAtTime(vol*.35,t+.25);
    g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
    var o1 = ac.createOscillator(), o2 = ac.createOscillator(), g2 = ac.createGain();
    o1.type='triangle'; o2.type='sine';
    o1.frequency.value = mtof(m); o2.frequency.value = mtof(m)*2; g2.gain.value = .22;
    o1.connect(g); o2.connect(g2); g2.connect(g); g.connect(master);
    o1.start(t); o2.start(t); o1.stop(t+dur+.05); o2.stop(t+dur+.05);
  }
  function scheduleBar(t, idx){
    var c = CHORDS[idx % CHORDS.length];
    note(c.bass, t, BAR*1.05, .20);
    ORDER.forEach(function(k,i){ note(c.arp[k] , t+i*EIGHTH, 1.6, .075); });
    c.mel.forEach(function(n){ note(n[1], t+n[0]*EIGHTH, n[2]*EIGHTH*1.5+.4, .11); });
  }
  function tick(){
    while(nextBar < ac.currentTime + 2.5){
      scheduleBar(nextBar, barIdx++);
      nextBar += BAR;
    }
  }
  function startMusic(){
    var AC = window.AudioContext || window.webkitAudioContext;
    if(!AC) return;
    if(!ac){
      ac = new AC();
      master = ac.createGain(); master.gain.value = 0;
      var rev = makeReverb(), wet = ac.createGain(); wet.gain.value = .55;
      var dly = ac.createDelay(1); dly.delayTime.value = EIGHTH*1.5;
      var fb = ac.createGain(); fb.gain.value = .28; dly.connect(fb); fb.connect(dly);
      master.connect(ac.destination);
      master.connect(rev); rev.connect(wet); wet.connect(ac.destination);
      master.connect(dly); dly.connect(wet);
    }
    if(ac.state === 'suspended') ac.resume();
    nextBar = ac.currentTime + .15; barIdx = 0;
    tick(); timer = setInterval(tick, 400);
    master.gain.cancelScheduledValues(ac.currentTime);
    master.gain.setValueAtTime(master.gain.value, ac.currentTime);
    master.gain.linearRampToValueAtTime(.8, ac.currentTime + 4);
    playing = true;
  }
  var soundBtn = $('#sound');
  soundBtn.addEventListener('click', function(){
    if(!ac) return;
    if(playing){
      master.gain.cancelScheduledValues(ac.currentTime);
      master.gain.setTargetAtTime(0, ac.currentTime, .15);
      setTimeout(function(){ if(!playing) ac.suspend(); }, 700);
      playing = false; soundBtn.classList.add('muted'); $('#soundTxt').textContent = 'Выкл.';
    } else {
      ac.resume(); master.gain.cancelScheduledValues(ac.currentTime);
      master.gain.setTargetAtTime(.8, ac.currentTime, .3);
      playing = true; soundBtn.classList.remove('muted'); $('#soundTxt').textContent = 'Музыка';
    }
  });
  document.addEventListener('visibilitychange', function(){
    if(!ac) return;
    if(document.hidden) ac.suspend(); else if(playing) ac.resume();
  });

  /* ---------- Открытие конверта ---------- */
  var seal = $('#seal'), env = $('#env'), intro = $('#intro'), stage = $('#stage'), opened = false;
  function openEnvelope(){
    if(opened) return; opened = true;
    try{ startMusic(); }catch(e){}
    intro.classList.add('opening');
    env.classList.add('open');                       // печать ломается, клапан открывается, письмо выезжает
    setTimeout(function(){ stage.classList.add('leave'); }, 2700);   // «въезжаем» в письмо
    setTimeout(function(){
      invite.classList.add('show');
      soundBtn.classList.add('vis');
    }, 3100);
    setTimeout(function(){
      intro.classList.add('gone');
      startObserver();
      sections[0].classList.add('in');               // тут начинается анимация: жених кружит невесту
    }, 3700);
  }
  seal.addEventListener('click', openEnvelope);
  env.addEventListener('click', function(e){ if(e.target === env || e.target.closest('.env-front,.flap,.letter')) openEnvelope(); });
})();
</script>
